import hashlib
import json
import os
import sys
from pathlib import Path


def pytest_sessionfinish(session, exitstatus):
    root = Path.cwd().resolve()
    modules = {}
    for name, module in sorted(sys.modules.items()):
        file = getattr(module, "__file__", None)
        if file and (name == "lobmm" or name.startswith("lobmm.")):
            path = Path(file).resolve()
            if not path.is_relative_to(root / "src"):
                raise RuntimeError("foreign suite module: " + name)
            modules[name] = {"path": str(path), "sha256": hashlib.sha256(path.read_bytes()).hexdigest()}
    Path(os.environ["LOB_COST_SUITE_IMPORTS"]).write_text(json.dumps({"exitstatus": int(exitstatus), "root": str(root), "modules": modules}, indent=2, sort_keys=True) + "\n")
