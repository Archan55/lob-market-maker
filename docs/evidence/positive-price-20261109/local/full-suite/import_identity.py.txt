import hashlib
import json
import pathlib
import sys


def pytest_collection_finish(session):
    root = pathlib.Path('/workspace/lob-market-maker-price-domain').resolve()
    found = {}
    for name, module in sorted(sys.modules.items()):
        if name == 'lobmm' or name.startswith('lobmm.'):
            filename = getattr(module, '__file__', None)
            if filename:
                path = pathlib.Path(filename).resolve()
                assert path.is_relative_to(root / 'src/lobmm'), str(path)
                found[name] = {'path': str(path), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}
    (root / 'experiments/positive-price-20261109/full-suite/imports.json').write_text(json.dumps(found, indent=2) + '\n')
