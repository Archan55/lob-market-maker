def pytest_collection_finish(session):
 root=pathlib.Path('/workspace/lob-market-maker-ordering-fix').resolve()
 found={}
 for name,m in sorted(sys.modules.items()):
  if name=='lobmm' or name.startswith('lobmm.'):
   f=getattr(m,'__file__',None)
   if f:

    assert p.is_relative_to(root/'src/lobmm'),str(p)
    found[name]={'path':str(p),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
 (root/'experiments/same-time-reporting-20261107/full-suite/imports.json').write_text(json.dumps(found,indent=2)+'\n')
